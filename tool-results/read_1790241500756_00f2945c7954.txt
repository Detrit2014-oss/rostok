   140→function makeEgg(index: number): Pet {
   141→  const species = PET_CATALOG[index % PET_CATALOG.length];
   142→  return {
   143→    id: `p${Date.now()}${Math.floor(Math.random() * 1000)}`,
   144→    name: species.name,
   145→    type: species.type,
   146→    bornAt: Date.now(),
   147→    growthMinutes: 0,
   148→    xp: 0,
   149→    frame: "none",
   150→  };
   151→}

   153→/** Боты живут своей жизнью: прогресс детерминирован сидом и днём недели. */
   154→function updateBots(c: Challenge): void {
   155→  const wd = new Date().getDay() === 0 ? 7 : new Date().getDay();
   156→  const rnd = seededRandom(stringHash(c.id));
   157→  for (const p of c.participants) {
   158→    if (p.isUser) continue;
   159→    const pacePerDay = 40 + Math.floor(rnd() * 60); // 40–99 мин/день
   160→    p.minutes = pacePerDay * wd + Math.floor(rnd() * 30);
   161→  }
   162→}

   340→        };
   341→        if (!wasAdult && petStage(updatedTarget) >= 3) {
   342→          evolvedName = updatedTarget.name;
   343→        }

   438→          questProgress = {};
   439→          questClaimed = [];
   440→        }

   443→        if (!q) return;

   445→        if (cur >= q.target) return;

   448→          questProgress: { ...questProgress, [questId]: cur + amount },

   459→        void claimed;

   746→export function daysLeft(): number {
   747→  const now = new Date();
   748→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   749→  const monday = new Date(today);
   750→  const wd = now.getDay() === 0 ? 7 : now.getDay();
   751→  monday.setDate(today.getDate() - (wd - 1));
   752→  const nextMonday = new Date(monday);
   753→  nextMonday.setDate(monday.getDate() + 7);
   754→  return Math.round((nextMonday.getTime() - today.getTime()) / 86400000);
   755→}

   757→/** Засчитанные секунды текущей сессии (фон + текущий период вне вкладки). */
   758→export function countedSecondsSoFar(s: {
   759→  sessionStartedAt: number | null;
   760→  countedSec: number;
   761→  awaySinceMs: number | null;
   762→  timeMachine: boolean;
   763→  nowMs: number;
   764→}): number {
   765→  if (!s.sessionStartedAt) return 0;
   766→  if (s.timeMachine) {
   767→    return Math.floor((s.nowMs - s.sessionStartedAt) / 1000) * 60;
   768→  }
   769→  const pending = s.awaySinceMs
   770→    ? Math.max(0, Math.floor((s.nowMs - s.awaySinceMs) / 1000))
   771→    : 0;
   772→  return s.countedSec + pending;
   773→}
