     8→/// Анимированная сцена «Ростка»: небо, солнце, облака, лужайка, погода
     9→/// (v1.7.0) и питомцы, нарисованные полностью процедурно (CustomPaint).
    10→///
    11→/// v1.8.0: настоящие лапы с пальцами (не «палки»), питомцы спят ЛЁЖА,
    12→/// водные жители живут в пруду по центру лужайки, декоративные рамки.
    13→class PetCanvas extends StatefulWidget {
    14→  const PetCanvas({
    15→    super.key,
    16→    required this.pets,
    17→    this.sleeping = false,
    18→    this.weather,
    19→    this.frame = 'none',
    20→  });
    21→
    22→  final List<Pet> pets;
    23→
    24→  /// true — идёт сессия детокса: питомцы спят ЛЁЖА и «копят» рост (zzz).
    25→  final bool sleeping;
    26→
    27→  /// Погодное состояние сцены: null|partly|cloudy|fog|rain|snow|thunder.
    28→  final String? weather;
    29→
    30→  /// Декоративная рамка из магазина: none|gold|neon|flower.
    31→  final String frame;
    32→
    33→  @override
    34→  State<PetCanvas> createState() => _PetCanvasState();
    35→}
    36→
    37→class _PetCanvasState extends State<PetCanvas>
    38→    with SingleTickerProviderStateMixin {
    39→  late final AnimationController _phase =
    40→      AnimationController(vsync: this, duration: const Duration(seconds: 4))
    41→        ..repeat();
    42→
    43→  @override
    44→  void dispose() {
    45→    _phase.dispose();
    46→    super.dispose();
    47→  }
    48→
    49→  @override
    50→  Widget build(BuildContext context) {
    51→    return AnimatedBuilder(
    52→      animation: _phase,
    53→      builder: (BuildContext context, Widget? _) {
    54→        return CustomPaint(
    55→          size: Size.infinite,
    56→          painter: _PetScenePainter(
    57→            pets: widget.pets,
    58→            phase: _phase.value,
    59→            sleeping: widget.sleeping,
    60→            weather: widget.weather,
    61→            frame: widget.frame,
    62→          ),
    63→        );
    64→      },
    65→    );
    66→  }
    67→}
    68→
    69→class _PetScenePainter extends CustomPainter {
    70→  _PetScenePainter({
    71→    required this.pets,
    72→    required this.phase,
    73→    required this.sleeping,
    74→    required this.weather,
    75→    required this.frame,
    76→  });
    77→
    78→  final List<Pet> pets;
    79→  final double phase;
    80→  final bool sleeping;
    81→  final String? weather;
    82→  final String frame;
    83→
    84→  bool get _dimSky =>
    85→      weather == 'cloudy' || weather == 'rain' || weather == 'thunder';
    86→  bool get _hasPond => pets.any((Pet p) => isAquatic(p.type));
    87→
    88→  @override
    89→  void paint(Canvas canvas, Size size) {
    90→    _paintSky(canvas, size);
    91→    _paintGround(canvas, size);
    92→    if (_hasPond) _paintPond(canvas, size);
    93→    _paintPets(canvas, size);
    94→    _paintWeather(canvas, size);
    95→    if (frame != 'none') _paintFrame(canvas, size);
    96→  }
    97→
    98→  // ── Небо и погода ────────────────────────────────────────────────────
    99→  void _paintSky(Canvas canvas, Size size) {
   100→    final Rect rect = Offset.zero & size;
   101→    final List<Color> colors = switch (weather) {
   102→      'thunder' => const <Color>[Color(0xFF6B7B95), Color(0xFFB9C6A8)],
   103→      'rain' => const <Color>[Color(0xFF8FB8CF), Color(0xFFCBE3C8)],
   104→      'cloudy' => const <Color>[Color(0xFFAFC6D6), Color(0xFFE2EFD8)],
   105→      'snow' => const <Color>[Color(0xFFBAD8EE), Color(0xFFEFF7EF)],
   106→      _ => const <Color>[Color(0xFFA6E4FF), Color(0xFFEAF9E0)],
   107→    };
   108→    canvas.drawRect(
   109→      rect,
   110→      Paint()
   111→        ..shader = LinearGradient(
   112→          begin: Alignment.topCenter,
   113→          end: Alignment.bottomCenter,
   114→          colors: colors,
   115→        ).createShader(rect),
   116→    );
   117→
   118→    // Солнце с гало — прячется при непогоде.
   119→    if (!_dimSky) {
   120→      final Offset sun = Offset(size.width * 0.16, size.height * 0.14);
   121→      canvas.drawCircle(sun, size.width * 0.085,
   122→          Paint()..color = const Color(0xFFFFC800).withOpacity(0.25));
   123→      canvas.drawCircle(sun, size.width * 0.055,
   124→          Paint()..color = const Color(0xFFFFC800));
   125→    }
   126→
   127→    // Облака: обычные две + доп. при частичной/пасмурной погоде.
   128→    _paintCloud(canvas, Offset(size.width * 0.55, size.height * 0.14),
   129→        size.width * 0.045);
   130→    _paintCloud(canvas, Offset(size.width * 0.78, size.height * 0.24),
   131→        size.width * 0.035);
   132→    if (weather == 'partly' || weather == 'cloudy' || weather == 'rain') {
   133→      _paintCloud(canvas, Offset(size.width * 0.32, size.height * 0.10),
   134→          size.width * 0.038);
   135→      _paintCloud(canvas, Offset(size.width * 0.92, size.height * 0.09),
   136→          size.width * 0.03);
   137→    }
   138→  }
   139→
   140→  void _paintCloud(Canvas canvas, Offset center, double r) {
   141→    final Paint p = Paint()..color = Colors.white.withOpacity(0.9);
   142→    canvas.drawCircle(center + Offset(-r * 0.8, r * 0.25), r * 0.65, p);
   143→    canvas.drawCircle(center, r, p);
   144→    canvas.drawCircle(center + Offset(r * 0.8, r * 0.25), r * 0.6, p);
   145→  }
   146→
   147→  void _paintWeather(Canvas canvas, Size size) {
   148→    switch (weather) {
   149→      case 'fog':
   150→        for (int i = 0; i < 3; i++) {
   151→          final double y =
   152→              size.height * (0.45 + 0.16 * i) + math.sin(phase * 6.28 + i) * 8;
   153→          final Rect band =
   154→              Rect.fromLTWH(-20 + math.sin(phase * 6.28 + i * 2) * 14, y,
   155→                  size.width + 40, size.height * 0.075);
   156→          canvas.drawRRect(
   157→            RRect.fromRectAndRadius(band, const Radius.circular(20)),
   158→            Paint()..color = Colors.white.withOpacity(0.38 - i * 0.07),
   159→          );
   160→        }
   161→        break;
   162→      case 'rain':
   163→        final Paint drop = Paint()
   164→          ..color = const Color(0xFF9FD4EF).withOpacity(0.75)
   165→          ..strokeWidth = 2.2
   166→          ..strokeCap = StrokeCap.round;
   167→        for (int i = 0; i < 34; i++) {
   168→          final double fx = ((i * 97) % 100) / 100;
   169→          final double fall = ((phase * 2.2 + fx * 3.0) % 1.0);
   170→          final double x = size.width * fx + fall * 18;
   171→          final double y = size.height * (fall - 0.08);
   172→          canvas.drawLine(Offset(x, y), Offset(x - 5, y + 16), drop);
   173→        }
   174→        break;
   175→      case 'snow':
   176→        final Paint flake = Paint()..color = Colors.white.withOpacity(0.9);
   177→        for (int i = 0; i < 26; i++) {
   178→          final double fx = ((i * 61) % 100) / 100;
   179→          final double fall = ((phase * 0.9 + fx * 2.4) % 1.0);
   180→          final double x = size.width * fx + math.sin(fall * 9 + i) * 14;
   181→          final double y = size.height * fall;
   182→          canvas.drawCircle(Offset(x, y), 2.4 + (i % 3), flake);
   183→        }
   184→        break;
   185→      case 'thunder':
   186→        // Тёмная вуаль.
   187→        canvas.drawRect(
   188→          Offset.zero & size,
   189→          Paint()..color = const Color(0xFF2E3A55).withOpacity(0.22),
   190→        );
   191→        // Вспышка молнии в узком окне цикла.
   192→        final double cyc = (phase * 1.35) % 1.0;
   193→        if (cyc < 0.09) {
   194→          final double flash = (1 - cyc / 0.09).clamp(0.0, 1.0);
   195→          canvas.drawRect(
   196→            Offset.zero & size,
   197→            Paint()..color = Colors.white.withOpacity(0.30 * flash),
   198→          );
   199→          final Paint bolt = Paint()
   200→            ..color = const Color(0xFFFFE45C)
   201→            ..style = PaintingStyle.stroke
   202→            ..strokeWidth = 4.5
   203→            ..strokeJoin = StrokeJoin.round
   204→            ..strokeCap = StrokeCap.round;
   205→          final double bx = size.width * 0.68;
   206→          final Path p = Path()
   207→            ..moveTo(bx, size.height * 0.06)
   208→            ..lineTo(bx - 26, size.height * 0.30)
   209→            ..lineTo(bx + 6, size.height * 0.32)
   210→            ..lineTo(bx - 20, size.height * 0.62);
   211→          canvas.drawPath(p, bolt);
   212→        }
   213→        break;
   214→      default:
   215→        break;
   216→    }
   217→  }
   218→
   219→  // ── Земля и пруд ─────────────────────────────────────────────────────
   220→  void _paintGround(Canvas canvas, Size size) {
   221→    final double groundY = size.height * 0.74;
   222→    final Rect rect =
   223→        Rect.fromLTWH(0, groundY, size.width, size.height - groundY);
   224→    canvas.drawRRect(
   225→      RRect.fromRectAndCorners(
   226→        rect,
   227→        topLeft: const Radius.circular(28),
   228→        topRight: const Radius.circular(28),
   229→      ),
   230→      Paint()..color = const Color(0xFF90D26D),
   231→    );
   232→
   233→    // Кустики для объёма
   234→    final Paint bush = Paint()..color = const Color(0xFF7FC45C);
   235→    canvas.drawCircle(
   236→        Offset(size.width * 0.12, groundY + size.height * 0.09), 22, bush);
   237→    canvas.drawCircle(
   238→        Offset(size.width * 0.90, groundY + size.height * 0.07), 18, bush);
   239→
   240→    // Цветочки
   241→    final List<Offset> flowers = <Offset>[
   242→      Offset(size.width * 0.06, groundY + size.height * 0.14),
   243→      Offset(size.width * 0.28, groundY + size.height * 0.10),
   244→      Offset(size.width * 0.72, groundY + size.height * 0.16),
   245→      Offset(size.width * 0.95, groundY + size.height * 0.12),
   246→    ];
   247→    final List<Color> fColors = <Color>[
   248→      const Color(0xFFFF8FB1),
   249→      const Color(0xFFFFD166),
   250→      const Color(0xFFEF476F),
   251→      const Color(0xFFFFD166),
   252→    ];
   253→    for (int i = 0; i < flowers.length; i++) {
   254→      canvas.drawCircle(
   255→          flowers[i], 4, Paint()..color = fColors[i % fColors.length]);
   256→      canvas.drawCircle(flowers[i], 1.6, Paint()..color = Colors.white);
   257→    }
   258→  }

   260→  /// Пруд по центру лужайки (v1.8.0) — дом водных питомцев.
   261→  void _paintPond(Canvas canvas, Size size) {
   262→    final double groundY = size.height * 0.74;
   263→    final Offset c = Offset(size.width * 0.5, groundY + size.height * 0.10);
   264→    final double rw = size.width * 0.30;
   265→    final double rh = size.height * 0.115;
   266→
   267→    // Песчаный берег.
   268→    canvas.drawOval(
   269→      Rect.fromCenter(center: c, width: rw * 2.16, height: rh * 2.16),
   270→      Paint()..color = const Color(0xFFE8D8A8),
   271→    );
   272→    // Вода.
   273→    canvas.drawOval(
   274→      Rect.fromCenter(center: c, width: rw * 2, height: rh * 2),
   275→      Paint()..color = const Color(0xFF6EC1E4),
   276→    );
   277→    // Блики-волны.
   278→    final Paint wave = Paint()
   279→      ..color = Colors.white.withOpacity(0.5)
   280→      ..style = PaintingStyle.stroke
   281→      ..strokeWidth = 2
   282→      ..strokeCap = StrokeCap.round;
   283→    for (int i = 0; i < 3; i++) {
   284→      final double wy = c.dy - rh * 0.4 + i * rh * 0.42;
   285→      final Rect wr = Rect.fromCenter(
   286→        center: Offset(c.dx + (i - 1) * rw * 0.18, wy),
   287→        width: rw * 0.5,
   288→        height: rh * 0.2,
   289→      );
   290→      canvas.drawArc(wr, 0.15 * math.pi, 0.7 * math.pi, false, wave);
   291→    }
   292→  }
   293→
   294→  /// Передняя кромка воды — перекрывает нижнюю часть тела водного питомца.
   295→  void _paintWaterFront(Canvas canvas, Offset pondC, double rw, double rh) {
   296→    canvas.drawOval(
   297→      Rect.fromCenter(center: pondC, width: rw * 2, height: rh * 2),
   298→      Paint()..color = const Color(0xFF6EC1E4).withOpacity(0.45),
   299→    );
   300→    // Круги на воде.
   301→    final Paint ripple = Paint()
   302→      ..color = Colors.white.withOpacity(0.55)
   303→      ..style = PaintingStyle.stroke
   304→      ..strokeWidth = 2
   305→      ..strokeCap = StrokeCap.round;
   306→    for (int i = 0; i < 2; i++) {
   307→      final double t = ((phase * 1.2 + i * 0.5) % 1.0);
   308→      canvas.drawOval(
   309→        Rect.fromCenter(
   310→          center: pondC + Offset(0, rh * 0.3),
   311→          width: rw * (0.7 + t * 0.9),
   312→          height: rh * (0.7 + t * 0.9) * 0.42,
   313→        ),
   314→        ripple..color = Colors.white.withOpacity(0.55 * (1 - t)),
   315→      );
   316→    }
   317→  }

   319→  // ── Питомцы ──────────────────────────────────────────────────────────
   320→  void _paintPets(Canvas canvas, Size size) {
   321→    if (pets.isEmpty) return;
   322→    final double groundY = size.height * 0.76;
   323→
   324→    // Сначала сухопутные, затем водные (они в пруду поверх воды).
   325→    final Offset pondC =
   326→        Offset(size.width * 0.5, size.height * 0.74 + size.height * 0.10);
   327→    final double prw = size.width * 0.30;
   328→    final double prh = size.height * 0.115;
   329→
   330→    final List<Pet> land =
   331→        pets.where((Pet p) => !isAquatic(p.type)).toList();
   332→    final List<Pet> water = pets.where((Pet p) => isAquatic(p.type)).toList();
   333→
   334→    for (int i = 0; i < land.length; i++) {
   335→      final Pet pet = land[i];
   336→      final double t = land.length == 1 ? 0.5 : 0.14 + 0.72 * i / (land.length - 1);
   337→      final double x = size.width * t;
   338→      final double bounce =
   339→          math.sin(phase * 2 * math.pi + i * 1.7) * (sleeping ? 0.5 : 2.5);
   340→      final Offset base = Offset(x, groundY - bounce);
   341→      _paintPet(canvas, base, pet, i, pondC: null);
   342→    }
   343→
   344→    for (int i = 0; i < water.length; i++) {
   345→      final Pet pet = water[i];
   346→      final double bounce = math.sin(phase * 2 * math.pi * 0.6 + i * 2.1) * 3;
   347→      // Водные питомцы плавают по центру пруда (можно несколько — в ряд).
   348→      final double dx =
   349→          water.length == 1 ? 0 : (i - (water.length - 1) / 2) * prw * 0.55;
   350→      final Offset base = Offset(pondC.dx + dx, pondC.dy - prh * 0.35 - bounce);
   351→      _paintPet(canvas, base, pet, i, pondC: Offset(pondC.dx, pondC.dy));
   352→    }
   353→
   354→    // Передняя вода поверх водных питомцев — тела «погружены».
   355→    if (water.isNotEmpty) _paintWaterFront(canvas, pondC, prw, prh);
   356→  }

   358→  Color _frameColor() => switch (frame) {
   359→        'gold' => const Color(0xFFFFC800),
   360→        'neon' => const Color(0xFF1CB0F6),
   361→        'flower' => const Color(0xFFFF8FB1),
   362→        _ => const Color(0x00000000),
   363→      };
   364→
   365→  void _paintFrame(Canvas canvas, Size size) {
   366→    final Color c = _frameColor();
   367→    if (c.a == 0) return;
   368→    final Rect r = (Offset.zero & size).deflate(7);
   369→    final Paint p = Paint()
   370→      ..style = PaintingStyle.stroke
   371→      ..strokeWidth = 5.5
   372→      ..color = c;
   373→    canvas.drawRRect(
   374→      RRect.fromRectAndRadius(r, const Radius.circular(24)),
   375→      p,
   376→    );
   377→    // Уголки-бусины.
   378→    final Paint bead = Paint()..color = c.withOpacity(0.85);
   379→    for (final Offset o in <Offset>[
   380→      Offset(r.left + 10, r.top + 10),
   381→      Offset(r.right - 10, r.top + 10),
   382→      Offset(r.left + 10, r.bottom - 10),
   383→      Offset(r.right - 10, r.bottom - 10),
   384→    ]) {
   385→      canvas.drawCircle(o, 4.5, bead);
   386→    }
   387→    if (frame == 'flower') {
   388→      for (final Offset o in <Offset>[
   389→        Offset(size.width * 0.5, r.top + 9),
   390→        Offset(size.width * 0.5, r.bottom - 9),
   391→      ]) {
   392→        canvas.drawCircle(o, 4, bead);
   393→        canvas.drawCircle(o, 1.8, Paint()..color = Colors.white);
   394→      }
   395→    }
   396→  }

   398→  /// Главный вход: рисует питомца (яйцо / зверя / растение).
   399→  /// [pondC] — центр пруда, если питомец водный.
   400→  void _paintPet(Canvas canvas, Offset base, Pet pet, int index,
   401→      {Offset? pondC}) {
   402→    final Color body = speciesBody(pet.type);
   403→    if (pet.stage == 0) {
   404→      _paintEgg(canvas, base, body, pet.stageProgress);
   405→      return;
   406→    }
   407→    if (isPlant(pet.type)) {
   408→      _paintPlant(canvas, base, pet);
   409→      return;
   410→    }
   411→
   412→    final SpeciesStyle st =
   413→        kSpeciesStyles[pet.type] ?? kSpeciesStyles[PetType.fox]!;
   414→    final Color belly = speciesBelly(pet.type);
   415→    final bool lying = sleeping; // v1.8.0: спим только ЛЁЖА.
   416→    final double scale = 0.55 + pet.stage * 0.22;
   417→    final double bw = 64 * scale;
   418→    final double bh = (lying ? 44 * 0.74 : 54) * scale;
   419→    final PetType type = pet.type;
   420→
   421→    canvas.save();
   422→    if (lying) {
   423→      // Лёжа: чуть сплющиваем и присыпаем «одеялом» тишины.
   424→      canvas.translate(base.dx, base.dy);
   425→      canvas.scale(1.0, 0.86);
   426→      canvas.translate(-base.dx, -base.dy);
   427→    }
   428→
   429→    final Offset bodyC = base - Offset(0, bh / 2 - (lying ? 2 : 0));
   430→    final Paint earPaint = Paint()..color = body;
   431→    final double earY = bodyC.dy - bh * 0.42;
   432→    final double legH = bh * 0.30;
   433→
   434→    // ── За телом: хвост, лапы (настоящие, с пальцами) ─────────────────
   435→    _paintTail(canvas, bodyC, bw, bh, st.tail, body, scale);
   436→
   437→    if (!lying && !isAquatic(type)) {
   438→      _paintLegs(canvas, base, bw, legH, body, belly, scale);
   439→    }
   440→
   441→    // ── Тело и животик ────────────────────────────────────────────────
   442→    canvas.drawOval(
   443→      Rect.fromCenter(center: bodyC, width: bw, height: bh),
   444→      Paint()..color = body,
   445→    );
   446→    canvas.drawOval(
   447→      Rect.fromCenter(
   448→        center: bodyC + Offset(0, bh * 0.18),
   449→        width: bw * 0.62,
   450→        height: bh * 0.5,
   451→      ),
   452→      Paint()..color = belly,
   453→    );
   454→
   455→    // ── Особые детали поверх тела ─────────────────────────────────────
   456→    _paintExtra(canvas, bodyC, bw, bh, st.extra, type, body, scale);
   457→
   458→    // ── Уши ───────────────────────────────────────────────────────────
   459→    _paintEars(canvas, bodyC, bw, bh, st.ear, body, belly, scale, earY);
   460→
   461→    // ── Лицо ──────────────────────────────────────────────────────────
   462→    _paintFace(canvas, bodyC, bw, bh, st, type, belly, scale, lying);
   463→
   464→    canvas.restore();
   465→
   466→    // zzz над спящим питомцем
   467→    if (sleeping && index == pets.length - 1) {
   468→      _paintZzz(canvas, Offset(bodyC.dx + bw * 0.55, bodyC.dy - bh * 0.7), scale);
   469→    }
   470→  }

   472→  // ── Части тела ───────────────────────────────────────────────────────
   473→  void _paintLegs(Canvas canvas, Offset base, double bw, double legH,
   474→      Color body, Color belly, double scale) {
   475→    final Paint leg = Paint()..color = body;
   476→    final Paint paw = Paint()..color = Color.alphaBlend(body.withOpacity(0.72), Colors.black12);
   477→    final Paint toe = Paint()..color = belly;
   478→    final double legW = bw * 0.15;
   479→    final double topY = base.dy - legH * 1.55;
   480→    for (final double sx in <double>[-0.30, 0.30]) {
   481→      final double x = base.dx + bw * sx;
   482→      // Нога — скруглённый столбик с лёгким сужением.
   483→      canvas.drawRRect(
   484→        RRect.fromRectAndRadius(
   485→          Rect.fromLTWH(x - legW / 2, topY, legW, legH * 1.55),
   486→          Radius.circular(legW * 0.45),
   487→        ),

   805→        break;

  1091→  void _paintPlant(Canvas canvas, Offset base, Pet pet) {

  1093→        kSpeciesStyles[pet.type] ?? kSpeciesStyles[PetType.cactus]!;

  1096→
  1097→    // Горшочек с грунтом.

  1099→      ..moveTo(-15 * scale, -26 * scale)

  1302→  void _paintEgg(Canvas canvas, Offset base, Color spotColor, double progress) {
  1303→    final double eggS = 0.95 + 0.45 * progress;
  1304→    final double w = 34 * eggS;
  1305→    final double h = 44 * eggS;
  1306→    final double wobble =
  1307→        math.sin(phase * 2 * math.pi * 1.6) * 0.06 * (0.35 + progress);
  1308→
  1309→    canvas.save();
  1310→    canvas.translate(base.dx, base.dy);
  1311→    canvas.rotate(wobble);
  1312→
  1313→    final Rect r =
  1314→        Rect.fromCenter(center: Offset(0, -h / 2), width: w, height: h);
  1315→    canvas.drawOval(r, Paint()..color = const Color(0xFFFFF6E3));
  1316→    canvas.drawOval(
  1317→        r.deflate(w * 0.12), Paint()..color = const Color(0xFFFFFBF0));
  1318→
  1319→    final Paint spots = Paint()..color = spotColor.withOpacity(0.65);
  1320→    canvas.drawCircle(Offset(-w * 0.18, -h * 0.55), 3.5 * eggS, spots);
  1321→    canvas.drawCircle(Offset(w * 0.15, -h * 0.35), 2.6 * eggS, spots);
  1322→    canvas.drawCircle(Offset(-w * 0.05, -h * 0.75), 2.0 * eggS, spots);
  1323→
  1324→    final Paint crack = Paint()
  1325→      ..color = const Color(0xFFC9A96E)
  1326→      ..style = PaintingStyle.stroke
  1327→      ..strokeWidth = 2.2
  1328→      ..strokeCap = StrokeCap.round
  1329→      ..strokeJoin = StrokeJoin.round;
  1330→    if (progress > 0.45) {
  1331→      canvas.drawPath(
  1332→        Path()
  1333→          ..moveTo(0, -h * 0.78)
  1334→          ..lineTo(w * 0.14, -h * 0.66)
  1335→          ..lineTo(-w * 0.07, -h * 0.54)
  1336→          ..lineTo(w * 0.1, -h * 0.44),
  1337→        crack,
  1338→      );
  1339→    }
  1340→    if (progress > 0.75) {
  1341→      canvas.drawPath(
  1342→        Path()
  1343→          ..moveTo(-w * 0.05, -h * 0.52)
  1344→          ..lineTo(-w * 0.2, -h * 0.4)
  1345→          ..lineTo(w * 0.02, -h * 0.28),
  1346→        crack,
  1347→      );
