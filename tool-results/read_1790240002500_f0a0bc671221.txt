    99→    99→  void _paintSky(Canvas canvas, Size size) {
   100→   100→    final Rect rect = Offset.zero & size;
   101→   101→    final List<Color> colors = switch (weather) {
   102→   102→      'thunder' => const <Color>[Color(0xFF6B7B95), Color(0xFFB9C6A8)],
   103→   103→      'rain' => const <Color>[Color(0xFF8FB8CF), Color(0xFFCBE3C8)],
   104→   104→      'cloudy' => const <Color>[Color(0xFFAFC6D6), Color(0xFFE2EFD8)],
   105→   105→      'snow' => const <Color>[Color(0xFFBAD8EE), Color(0xFFEFF7EF)],
   106→   106→      _ => const <Color>[Color(0xFFA6E4FF), Color(0xFFEAF9E0)],
   107→   107→    };
   108→   108→    canvas.drawRect(
   109→   109→      rect,
   110→   110→      Paint()
   111→   111→        ..shader = LinearGradient(
   112→   112→          begin: Alignment.topCenter,
   113→   113→          end: Alignment.bottomCenter,
   114→   114→          colors: colors,
   115→   115→        ).createShader(rect),
   116→   116→    );
   117→   117→
   118→   118→    // Солнце с гало — прячется при непогоде.
   119→   119→    if (!_dimSky) {
   120→   120→      final Offset sun = Offset(size.width * 0.16, size.height * 0.14);
   121→   121→      canvas.drawCircle(sun, size.width * 0.085,
   122→   122→          Paint()..color = const Color(0xFFFFC800).withOpacity(0.25));
   123→   123→      canvas.drawCircle(sun, size.width * 0.055,
   124→   124→          Paint()..color = const Color(0xFFFFC800));
   125→   125→    }
   126→   126→
   127→   127→    // Облака: обычные две + доп. при частичной/пасмурной погоде.
   128→   128→    _paintCloud(canvas, Offset(size.width * 0.55, size.height * 0.14),
   129→   129→        size.width * 0.045);
   130→   130→    _paintCloud(canvas, Offset(size.width * 0.78, size.height * 0.24),
   131→   131→        size.width * 0.035);
   132→   132→    if (weather == 'partly' || weather == 'cloudy' || weather == 'rain') {
   133→   133→      _paintCloud(canvas, Offset(size.width * 0.32, size.height * 0.10),
   134→   134→          size.width * 0.038);
   135→   135→      _paintCloud(canvas, Offset(size.width * 0.92, size.height * 0.09),

   141→   141→    final Paint p = Paint()..color = Colors.white.withOpacity(0.9);

   147→   147→  void _paintWeather(Canvas canvas, Size size) {
   148→   148→    switch (weather) {
   149→   149→      case 'fog':
   150→   150→        for (int i = 0; i < 3; i++) {
   151→   151→          final double y =
   152→   152→              size.height * (0.45 + 0.16 * i) + math.sin(phase * 6.28 + i) * 8;
   153→   153→          final Rect band =
   154→   154→              Rect.fromLTWH(-20 + math.sin(phase * 6.28 + i * 2) * 14, y,
   155→   155→                  size.width + 40, size.height * 0.075);
   156→   156→          canvas.drawRRect(
   157→   157→            RRect.fromRectAndRadius(band, const Radius.circular(20)),
   158→   158→            Paint()..color = Colors.white.withOpacity(0.38 - i * 0.07),
   159→   159→          );
   160→   160→        }
   161→   161→        break;
   162→   162→      case 'rain':
   163→   163→        final Paint drop = Paint()
   164→   164→          ..color = const Color(0xFF9FD4EF).withOpacity(0.75)
   165→   165→          ..strokeWidth = 2.2
   166→   166→          ..strokeCap = StrokeCap.round;
   167→   167→        for (int i = 0; i < 34; i++) {
   168→   168→          final double fx = ((i * 97) % 100) / 100;
   169→   169→          final double fall = ((phase * 2.2 + fx * 3.0) % 1.0);
   170→   170→          final double x = size.width * fx + fall * 18;
   171→   171→          final double y = size.height * (fall - 0.08);
   172→   172→          canvas.drawLine(Offset(x, y), Offset(x - 5, y + 16), drop);
   173→   173→        }
   174→   174→        break;
   175→   175→      case 'snow':
   176→   176→        final Paint flake = Paint()..color = Colors.white.withOpacity(0.9);
   177→   177→        for (int i = 0; i < 26; i++) {
   178→   178→          final double fx = ((i * 61) % 100) / 100;
   179→   179→          final double fall = ((phase * 0.9 + fx * 2.4) % 1.0);
   180→   180→          final double x = size.width * fx + math.sin(fall * 9 + i) * 14;
   181→   181→          final double y = size.height * fall;
   182→   182→          canvas.drawCircle(Offset(x, y), 2.4 + (i % 3), flake);
   183→   183→        }
   184→   184→        break;
   185→   185→      case 'thunder':
   186→   186→        // Тёмная вуаль.
   187→   187→        canvas.drawRect(
   188→   188→          Offset.zero & size,
   189→   189→          Paint()..color = const Color(0xFF2E3A55).withOpacity(0.22),
   190→   190→        );
   191→   191→        // Вспышка молнии в узком окне цикла.
   192→   192→        final double cyc = (phase * 1.35) % 1.0;
   193→   193→        if (cyc < 0.09) {
   194→   194→          final double flash = (1 - cyc / 0.09).clamp(0.0, 1.0);
   195→   195→          canvas.drawRect(
   196→   196→            Offset.zero & size,
   197→   197→            Paint()..color = Colors.white.withOpacity(0.30 * flash),
   198→   198→          );
   199→   199→          final Paint bolt = Paint()
   200→   200→            ..color = const Color(0xFFFFE45C)
   201→   201→            ..style = PaintingStyle.stroke
   202→   202→            ..strokeWidth = 4.5
   203→   203→            ..strokeJoin = StrokeJoin.round
   204→   204→            ..strokeCap = StrokeCap.round;
   205→   205→          final double bx = size.width * 0.68;
   206→   206→          final Path p = Path()
   207→   207→            ..moveTo(bx, size.height * 0.06)
   208→   208→            ..lineTo(bx - 26, size.height * 0.30)
   209→   209→            ..lineTo(bx + 6, size.height * 0.32)
   210→   210→            ..lineTo(bx - 20, size.height * 0.62);
   211→   211→          canvas.drawPath(p, bolt);
   212→   212→        }
   213→   213→        break;
   214→   214→      default:
   215→   215→        break;
   216→   216→    }
   217→   217→  }

   320→   320→  void _paintPets(Canvas canvas, Size size) {
   321→   321→    if (pets.isEmpty) return;
   322→   322→    final double groundY = size.height * 0.76;
   323→   323→
   324→   324→    // Сначала сухопутные, затем водные (они в пруду поверх воды).
   325→   325→    final Offset pondC =
   326→   326→        Offset(size.width * 0.5, size.height * 0.74 + size.height * 0.10);
   327→   327→    final double prw = size.width * 0.30;
   328→   328→    final double prh = size.height * 0.115;
   329→   329→
   330→   330→    final List<Pet> land =
   331→   331→        pets.where((Pet p) => !isAquatic(p.type)).toList();
   332→   332→    final List<Pet> water = pets.where((Pet p) => isAquatic(p.type)).toList();
   333→   333→
   334→   334→    for (int i = 0; i < land.length; i++) {
   335→   335→      final Pet pet = land[i];
   336→   336→      final double t = land.length == 1 ? 0.5 : 0.14 + 0.72 * i / (land.length - 1);
   337→   337→      final double x = size.width * t;
   338→   338→      final double bounce =
   339→   339→          math.sin(phase * 2 * math.pi + i * 1.7) * (sleeping ? 0.5 : 2.5);
   340→   340→      final Offset base = Offset(x, groundY - bounce);
   341→   341→      _paintPet(canvas, base, pet, i, pondC: null);
   342→   342→    }
   343→   343→
   344→   344→    for (int i = 0; i < water.length; i++) {
   345→   345→      final Pet pet = water[i];
   346→   346→      final double bounce = math.sin(phase * 2 * math.pi * 0.6 + i * 2.1) * 3;
   347→   347→      // Водные питомцы плавают по центру пруда (можно несколько — в ряд).
   348→   348→      final double dx =
   349→   349→          water.length == 1 ? 0 : (i - (water.length - 1) / 2) * prw * 0.55;
   350→   350→      final Offset base = Offset(pondC.dx + dx, pondC.dy - prh * 0.35 - bounce);
   351→   351→      _paintPet(canvas, base, pet, i, pondC: Offset(pondC.dx, pondC.dy));
   352→   352→    }
   353→   353→
   354→   354→    // Передняя вода поверх водных питомцев — тела «погружены».
   355→   355→    if (water.isNotEmpty) _paintWaterFront(canvas, pondC, prw, prh);
   356→   356→  }

   399→   399→  /// [pondC] — центр пруда, если питомец водный.
   400→   400→  void _paintPet(Canvas canvas, Offset base, Pet pet, int index,
   401→   401→      {Offset? pondC}) {
   402→   402→    final Color body = speciesBody(pet.type);
   403→   403→    if (pet.stage == 0) {
   404→   404→      _paintEgg(canvas, base, body, pet.stageProgress);
   405→   405→      return;
   406→   406→    }
   407→   407→    if (isPlant(pet.type)) {
   408→   408→      _paintPlant(canvas, base, pet);
   409→   409→      return;
   410→   410→    }
   411→   411→
   412→   412→    final SpeciesStyle st =
   413→   413→        kSpeciesStyles[pet.type] ?? kSpeciesStyles[PetType.fox]!;
   414→   414→    final Color belly = speciesBelly(pet.type);
   415→   415→    final bool lying = sleeping; // v1.8.0: спим только ЛЁЖА.
   416→   416→    final double scale = 0.55 + pet.stage * 0.22;
   417→   417→    final double bw = 64 * scale;
   418→   418→    final double bh = (lying ? 44 * 0.74 : 54) * scale;
   419→   419→    final PetType type = pet.type;
   420→   420→
   421→   421→    canvas.save();
   422→   422→    if (lying) {
   423→   423→      // Лёжа: чуть сплющиваем и присыпаем «одеялом» тишины.
   424→   424→      canvas.translate(base.dx, base.dy);
   425→   425→      canvas.scale(1.0, 0.86);
   426→   426→      canvas.translate(-base.dx, -base.dy);
   427→   427→    }
   428→   428→
   429→   429→    final Offset bodyC = base - Offset(0, bh / 2 - (lying ? 2 : 0));
   430→   430→    final Paint earPaint = Paint()..color = body;
   431→   431→    final double earY = bodyC.dy - bh * 0.42;
   432→   432→    final double legH = bh * 0.30;
   433→   433→
   434→   434→    // ── За телом: хвост, лапы (настоящие, с пальцами) ─────────────────
   435→   435→    _paintTail(canvas, bodyC, bw, bh, st.tail, body, scale);
   436→   436→
   437→   437→    if (!lying && !isAquatic(type)) {
   438→   438→      _paintLegs(canvas, base, bw, legH, body, belly, scale);
   439→   439→    }
   440→   440→
   441→   441→    // ── Тело и животик ────────────────────────────────────────────────
   442→   442→    canvas.drawOval(
   443→   443→      Rect.fromCenter(center: bodyC, width: bw, height: bh),
   444→   444→      Paint()..color = body,
   445→   445→    );
   446→   446→    canvas.drawOval(
   447→   447→      Rect.fromCenter(
   448→   448→        center: bodyC + Offset(0, bh * 0.18),
   449→   449→        width: bw * 0.62,
   450→   450→        height: bh * 0.5,
   451→   451→      ),
   452→   452→      Paint()..color = belly,
   453→   453→    );
   454→   454→
   455→   455→    // ── Особые детали поверх тела ─────────────────────────────────────
   456→   456→    _paintExtra(canvas, bodyC, bw, bh, st.extra, type, body, scale);
   457→   457→
   458→   458→    // ── Уши ───────────────────────────────────────────────────────────
   459→   459→    _paintEars(canvas, bodyC, bw, bh, st.ear, body, belly, scale, earY);
   460→   460→
   461→   461→    // ── Лицо ──────────────────────────────────────────────────────────
   462→   462→    _paintFace(canvas, bodyC, bw, bh, st, type, belly, scale, lying);
